#include <iostream>
#include <algorithm>
#include <cstring>
#include <queue>
#include <unordered_map>

using namespace std;

typedef pair<int, string> PIS;

unordered_map<string, int> dist;
unordered_map<string, pair<char, string>> pre;

int dx[4] = {-1, 0, 1, 0}, dy[4] = {0, 1, 0, -1};
char cs[] = "urdl";

int f(string s)
{
    int res = 0;
    for (int i = 0; i < s.size(); i ++)
    {
        if (s[i] == 'x') continue;
        int a = i / 3, b = i % 3;
        int x = (s[i] - '1') / 3, y = (s[i] - '1') % 3;
        res += abs(a - x) + abs(b - y);
    }
    return res;
}

void astar(string start, string end)
{
    priority_queue<PIS, vector<PIS>, greater<PIS>> q;
    q.push({f(start), start});
    dist[start] = 0;

    while(q.size())
    {
        auto t = q.top();
        q.pop();

        if (t.second == end) break;
        string s = t.second;
        int sx, sy;
        for (int i = 0; i < s.size(); i ++)
        {
            if (s[i] == 'x')
            {
                sx = i / 3;
                sy = i % 3;
                break;
            }
        }
        
        for (int i = 0; i < 4; i ++)
        {
            int a = sx + dx[i], b = sy + dy[i];
            if (a < 0 || a >= 3 || b < 0 || b >= 3) continue;

            string tmp = s;
            swap(tmp[a * 3 + b], tmp[sx * 3 + sy]);
            if (!dist.count(tmp) || dist[tmp] > dist[s] + 1)
            {
                dist[tmp] = dist[s] + 1;
                q.push({f(tmp) + dist[tmp], tmp});
                pre[tmp] = {cs[i], s};
            }
        }
    }
}

int main()
{
    string start, st;

    for (int i = 1; i <= 9; i ++)
    {
        char x;
        cin >> x;
        start += x;
        if (x != 'x') st += x;
    }
    
    string end = "12345678x";

    int cnt = 0;
    for (int i = 0; i < st.size(); i ++)
        for (int j = i; j < st.size(); j ++)
            if (st[i] > st[j]) cnt ++;

    if (cnt % 2 == 0)
    {
        astar(start, end);
        vector<char> ans;
        while (end != start)
        {
            ans.push_back(pre[end].first);
            end = pre[end].second;
        }
        reverse(ans.begin(), ans.end());
        for (auto c : ans) cout << c ;
    } 
    else cout << "unsolvable" << endl;
    return 0;
}
